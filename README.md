```sql
SELECT 
    customer_id,
    COUNT(DISTINCT order_id) as total_orders,
    SUM(amount) as total_spent,
    CASE
        WHEN SUM(amount) > 1000 THEN 'Premium'
        WHEN SUM(amount) > 500 THEN 'Regular'
        ELSE 'Basic'
    END as customer_segment
FROM orders
GROUP BY customer_id
ORDER BY total_spent DESC;
```
